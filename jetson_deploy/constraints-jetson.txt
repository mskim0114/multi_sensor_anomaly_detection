# JETSON-RUNTIME version constraints.
#
# Purpose: pin the dependency versions that were actually verified on this
# Jetson, including transitive ones, so that a rebuild resolves to the same
# tree. Pair it with requirements-jetson.txt:
#
#   -r jetson_deploy/requirements-jetson.txt
#   -c jetson_deploy/constraints-jetson.txt
#
# A constraints file does NOT install anything by itself. It only bounds
# versions that the requirements pull in.
#
# Only pin what has been verified. An unverified pin is worse than no pin,
# because it turns a working resolution into a hard failure.
#
# Verified on: L4T R36.5.0, Ubuntu 22.04.5, Python 3.10.12, aarch64.

# --- The critical constraint -------------------------------------------------
# numpy MUST stay on 1.x: JetPack OpenCV / APT pandas / APT PIL are built
# against the numpy 1.x C ABI. See requirements-jetson.txt for the full reason.
numpy==1.26.4

# --- Inference runtime ------------------------------------------------------
# onnxruntime-gpu is intentionally NOT pinned here. It is supplied by
# requirements-jetson.txt as a hashed direct URL, and pip rejects a version
# constraint applied to a direct-URL requirement. The digest in that file is the
# stronger guarantee:
#   onnxruntime_gpu-1.24.0-cp310-cp310-linux_aarch64.whl
#   sha256 d980b934b9a29c1a9d6f39751edd7662b69fadd75556a10ff363773a58ce0950
